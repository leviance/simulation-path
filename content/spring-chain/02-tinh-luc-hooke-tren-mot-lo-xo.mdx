# Tính lực Hooke trên một lò xo

## Vấn đề cần giải quyết

Ta đã có topology nhưng chưa có quy luật khiến hai đầu kéo nhau. Trước khi tính lực cho cả dây, hãy chỉ giữ hai khối để trả lời ba câu hỏi: lò xo đang dài hay ngắn hơn tự nhiên, lực trên A hướng đi đâu, và lực trên B có đối nhau không?

## Hiểu ý tưởng trước khi viết code

Gọi A và B là hai đầu. Chọn hướng n **từ A sang B**:

$$
\delta=p_B-p_A,\quad d=|\delta|,\quad n=\delta/d,\quad x=d-L_0.
$$

Với quy ước này, lực trên A là $F_A=kxn$, còn $F_B=-F_A$. Khi x dương, A bị kéo về B; khi x âm, A bị đẩy ra xa B. Dấu trừ trong cách viết Hooke một chiều phụ thuộc cách chọn độ dời, không nên chép nó vào mọi vector formula.

Ví dụ L₀=1 m, d=1,2 m, k=100 N/m cho độ lớn 20 N. **Canvas của bài đang dùng k=120 N/m**, nên cùng khoảng cách cho 24 N. Góc thay đổi vector lực nhưng không đổi độ lớn khi d giữ nguyên.

### Bước 1 — Tách kết quả đo khỏi thao tác cộng lực

Thêm SpringSample sau hàm tạo chuỗi. Hàm ở bài này chỉ đọc state và trả kết quả; nó chưa đổi force, velocity hoặc position:

```cpp title="include/lab.hpp — SpringSample"

```

### Bước 2 — Viết lực theo quy ước đã chọn

Thêm hàm sau vào lab.hpp:

```cpp title="include/lab.hpp — sampleElasticSpring"
inline SpringSample sampleElasticSpring(const SpringChain& chain, const Spring& spring) {
    if (!validSpring(spring, chain.particles.size())) {
        return {};
    }
    const Particle& first = chain.particles[spring.first];
    const Particle& second = chain.particles[spring.second];
    if (!finiteParticle(first) || !finiteParticle(second)) {
        return {};
    }
    const Vec2 delta = subtract(second.position, first.position);
    const double distance = length(delta);
    if (!std::isfinite(distance) || distance <= kSpringEpsilon) {
        return {};
    }
    const Vec2 direction = scale(delta, 1.0 / distance);
    const double stretch = distance - spring.restLength;
    const Vec2 relativeVelocity = subtract(second.velocity, first.velocity);
    const double relativeSpeed = dot(relativeVelocity, direction);
    const double forceMagnitude = spring.stiffness * stretch;
    const Vec2 force = scale(direction, forceMagnitude);
    if (!finiteVec2(force) || !std::isfinite(relativeSpeed)) {
        return {};
    }
    return {true, direction, distance, stretch, relativeSpeed, force};
}
```

Distance và restLength là hai đại lượng khác nhau. RelativeSpeed đã được đo để dùng ở bài damping, nhưng chưa được cộng vào lực.

Nếu hai tâm trùng nhau, direction không xác định. Sample báo không hợp lệ; ở bước mô phỏng sau, caller sẽ từ chối cả bước thay vì bí mật bỏ liên kết ấy.

### Bước 3 — Đổi Scene thành cặp thử nghiệm

Thay rebuildScene bằng phiên bản này. Nhánh cặp chỉ phục vụ bài 2:

```cpp title="include/scene.hpp — rebuildScene"
inline void rebuildScene(Scene& scene) {
    stopScene(scene);
    scene.message.clear();
    scene.chain = makeVerticalChain(static_cast<std::size_t>(scene.count), {0.0, 0.0}, scene.restLength, scene.mass, 0.10, scene.stiffness, scene.damping);
    scene.chain = makeVerticalChain(2, {0.0, 0.0}, 1.0, scene.mass, 0.10, scene.stiffness, scene.damping);
    scene.selected = 1;
    scene.chain.particles[1].position = {scene.pairDistance * std::cos(scene.pairAngle), scene.pairDistance * std::sin(scene.pairAngle)};
    beginMeasurement(scene);
}
```

Cập nhật drawChain và renderScene theo phiên bản bài này để màu thể hiện stretch và hai mũi tên đọc cùng một sample:

```cpp title="include/view.hpp — drawChain"
inline void drawChain(Framebuffer& image, const Scene& scene, const WorldView& view) {
    for (const auto& spring : scene.chain.springs) {
        const auto& a = scene.chain.particles[spring.first];
        const auto& b = scene.chain.particles[spring.second];
        std::uint32_t color = 0x66738cffU;
        const double stretch = length(subtract(b.position, a.position)) - spring.restLength;
        if (stretch > 0.002) {
            color = 0xe5c07bffU;
        } else if (stretch < -0.002) {
            color = 0x56b6c2ffU;
        }
        drawCoil(image, worldToScreen(a.position, view), worldToScreen(b.position, view), color);
    }
    for (std::size_t i = 0; i < scene.chain.particles.size(); ++i) {
        const auto& particle = scene.chain.particles[i];
        const Vec2 center = worldToScreen(particle.position, view);
        if (i == scene.selected) {
            drawNode(image, center, std::max(5.0, particle.radius * view.scale + 3.0), 0xffffffffU);
        }
        drawNode(image, center, particle.radius * view.scale, i == 0 ? 0xc678ddffU : 0x53f0aeffU);
    }
}
```

```cpp title="include/view.hpp — renderScene"
inline void renderScene(Framebuffer& image, const Scene& scene) {
    image.clear();
    const WorldView view = makeWorldView(scene);
    rectangle(image, 40, 100, 690, 650, kRail, false);
    if (scene.debug) {
        for (int row = 1; row <= 10; ++row) {
            const double y = 100.0 + 50.0 * double(row);
            drawLine(image, 40, y, 690, y, kGrid);
        }
    }
    drawChain(image, scene, view);
    const Vec2 anchor = worldToScreen(scene.chain.anchorPosition, view);
    rectangle(image, int(anchor.x - 20), int(anchor.y - 25), int(anchor.x + 20), int(anchor.y - 10), 0xc678ddffU, true);
    drawLine(image, anchor.x, anchor.y - 10, anchor.x, anchor.y, 0xc678ddffU);
    const auto pair = sampleElasticSpring(scene.chain, scene.chain.springs.front());
    const auto a = scene.chain.particles[0].position;
    const auto b = scene.chain.particles[1].position;
    drawForceArrow(image, worldToScreen(a, view), worldToScreen(add(a, scale(pair.forceOnFirst, 0.08)), view));
    drawForceArrow(image, worldToScreen(b, view), worldToScreen(subtract(b, scale(pair.forceOnFirst, 0.08)), view));
}
```

Trong presentation.hpp, thay drawHud để hiện khoảng cách, độ giãn và số N. Các helper vẽ node/coil/arrow đã nằm trong Starter; không cần cài lại SDL resources:

```cpp title="include/presentation.hpp — drawHud"
inline bool drawHud(SDL_Renderer* renderer, const Scene& scene) {
    SDL_SetRenderDrawColor(renderer, 205, 215, 235, 255);
    char line[200]{};
    std::snprintf(line, sizeof(line), "SPRING CHAIN | %zu masses / %zu springs | t=%.3f s | %s", scene.chain.particles.size(), scene.chain.springs.size(), scene.elapsed, scene.paused ? "PAUSED" : "RUNNING");
    if (!SDL_RenderDebugText(renderer, 30, 20, line)) {
        return false;
    }
    if (!SDL_RenderDebugText(renderer, 30, 42, "Space pause | N step (lesson 4+) | R reset | F clean/debug | Esc quit")) {
        return false;
    }
    SDL_RenderDebugText(renderer, 30, 64, "Left/Right: distance | Up/Down: angle. F = k * (distance - restLength) * direction.");
    const auto sample = sampleElasticSpring(scene.chain, scene.chain.springs.front());
    std::snprintf(line, sizeof(line), "d=%.3f m | L0=1 m | stretch=%.3f m | F_A=(%.3f, %.3f) N", sample.distance, sample.stretch, sample.forceOnFirst.x, sample.forceOnFirst.y);
    SDL_RenderDebugText(renderer, 30, 700, line);
    if (!scene.message.empty()) {
        SDL_RenderDebugText(renderer, 30, 790, scene.message.c_str());
    }
    return true;
}
```


## Ghép các phần lại

Build Starter. Trái/phải đổi d; lên/xuống đổi góc. Với d=1,2 m và góc=0, F_A phải là (24,0) N. Giảm d xuống 0,8 m để thấy lực trên A đổi thành (−24,0) N. Xoay tới gần π/2, thành phần lực phải chuyển từ X sang Y.

Hai khối không chạy vì ta vẫn chưa tích phân. Đó là chủ đích: trước hết xác nhận hàm lực không thay đổi state.

## Thử làm sai để hiểu đúng

Chỉ đổi một chi tiết mỗi lần. Hãy dự đoán, chạy lại, ghi điều quan sát được rồi hoàn tác trước thí nghiệm tiếp theo.

1. Đổi stretch thành L₀−d nhưng giữ nguyên công thức F_A. Lò xo giãn sẽ đẩy hai đầu ra xa nhau.

2. Dùng delta thay direction mà vẫn nhân k·stretch. Độ lớn lực bị nhân thêm distance và sai đơn vị.

3. Cập nhật restLength=d sau mỗi lần kéo. Lò xo mất biến dạng nên không còn lực phục hồi.

## Tự kiểm tra kết quả

- Kéo giãn và nén cùng 0,2 m tạo lực đối dấu, cùng độ lớn.
- F_B=−F_A; xoay cặp không làm đổi độ lớn lực.
- Sample không sửa bất kỳ khối nào.
- Hai đầu trùng nhau được báo không hợp lệ, không sinh NaN.
