cmake_minimum_required(VERSION 3.24)
project(voxel_editorial_capture LANGUAGES CXX)

# This tool belongs to the website, never to a student's checkpoint or ZIP.
function(capture_checkpoint id slug stage)
    set(snapshot "${CMAKE_CURRENT_SOURCE_DIR}/../../examples/project-${id}-${slug}/checkpoints/0${stage}/include")
    if(NOT EXISTS "${snapshot}/trail/scene.hpp")
        message(FATAL_ERROR "Generate Project ${id} checkpoints first: ${snapshot}")
    endif()
    add_executable(take_${id}_${stage} record.cpp)
    target_compile_features(take_${id}_${stage} PRIVATE cxx_std_20)
    target_include_directories(take_${id}_${stage} PRIVATE "${snapshot}")
    target_compile_definitions(take_${id}_${stage} PRIVATE PROJECT_ID=${id} STAGE=${stage})
    if(MSVC)
        target_compile_options(take_${id}_${stage} PRIVATE /utf-8)
    endif()
endfunction()

capture_checkpoint(61 voxel-greedy-meshing 2)
capture_checkpoint(61 voxel-greedy-meshing 3)

capture_checkpoint(63 voxel-procedural-terrain 3)
